# Oracle driver — install this file in addition to requirements.txt when
# running the backend outside its Docker image. The Dockerfile installs it and
# validates thick-mode initialization during the image build.
#
# Installation:
#   pip install -r requirements.txt -r requirements-oracle.txt
#
# python-oracledb 4.x requires Oracle Client 19 or newer for thick mode. The
# Docker image therefore bundles Instant Client 19.32, which can connect to
# Oracle Database 11.2 and newer. The high-level API used here (connect,
# cursor, execute, init_oracle_client, and call_timeout) is stable in 4.x.
oracledb>=4.0.2,<5
